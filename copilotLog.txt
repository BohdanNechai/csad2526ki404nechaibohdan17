on:
  push:
    branches:
      - '*develop*'
      - '*master*'
  pull_request:

      - '*develop*'
      - '*master*'

jobs:
  cross_build:
    name: Cross-platform build (${{ matrix.os }})
    runs-on: ${{ matrix.os }}
    strategy:
      fail-fast: false
      matrix:
        os:
          - ubuntu-latest
          - windows-latest
          - macos-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Install CMake (Linux)
        if: runner.os == 'Linux'
        run: |
          sudo apt-get update
          sudo apt-get install -y cmake

      - name: Install CMake (macOS)
        if: runner.os == 'macOS'
        run: |
          # ensure Homebrew exists, then install cmake
          if ! command -v brew >/dev/null 2>&1; then
            /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
          fi
          brew update
          brew install cmake

      - name: Install CMake (Windows)
        if: runner.os == 'Windows'
        shell: powershell
        run: |
          # use Chocolatey to install CMake on Windows runners
          if (-not (Get-Command choco -ErrorAction SilentlyContinue)) {
            Write-Host "Chocolatey not found; attempting to install Chocolatey..."
            Set-ExecutionPolicy Bypass -Scope Process -Force
            [System.Net.ServicePointManager]::SecurityProtocol = [System.Net.ServicePointManager]::SecurityProtocol -bor 3072
            iex ((New-Object System.Net.WebClient).DownloadString('https://chocolatey.org/install.ps1'))
          }
          choco install cmake --no-progress -y

      - name: Configure (Linux / macOS)
        if: runner.os != 'Windows'
        run: |
          mkdir -p build
          cd build
          cmake .. -DCMAKE_BUILD_TYPE=Release

      - name: Configure (Windows)
        if: runner.os == 'Windows'
        shell: powershell
        run: |
          New-Item -ItemType Directory -Force -Path build | Out-Null
          Set-Location build
          cmake .. -DCMAKE_BUILD_TYPE=Release

      - name: Build (Linux / macOS)
        if: runner.os != 'Windows'
        run: |
          cd build
          cmake --build . --config Release -- -j$(nproc)

      - name: Build (Windows)
        if: runner.os == 'Windows'
        shell: powershell
        run: |
          Set-Location build
          cmake --build . --config Release -- /m

      - name: Upload build artifact
        uses: actions/upload-artifact@v4
        with:
          name: build-${{ matrix.os }}
          path: build

      - name: Deploy (Linux / macOS)
        if: runner.os != 'Windows'
        run: |

          }
